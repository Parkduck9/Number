function crc(buf) {
  let c = 0xffffffff;
  for (const v of buf) { c ^= v; for (let k=0;k<8;k++) c=(c>>>1)^((c&1)?0xedb88320:0); }
  return (c^0xffffffff)>>>0;
}
function chunk(tag,data) {
  const t=Buffer.from(tag), n=Buffer.alloc(4), c=Buffer.alloc(4);
  n.writeUInt32BE(data.length); c.writeUInt32BE(crc(Buffer.concat([t,data])));
  return Buffer.concat([n,t,data,c]);
}
function canvas(w,h,color) {
  const im={w,h,p:Buffer.alloc(w*h*4)};
  if(color) for(let i=0;i<im.p.length;i+=4) im.p.set(color,i);
  return im;
}
function readPNG(filename) {
  const data=fs.readFileSync(filename), parts=[];
  let pos=8,w,h,channels;
  assert.equal(data.subarray(0,8).toString('hex'),'89504e470d0a1a0a');
  while(pos<data.length) {
    const n=data.readUInt32BE(pos), tag=data.toString('ascii',pos+4,pos+8);
    const b=data.subarray(pos+8,pos+8+n);
    if(tag==='IHDR') {
      w=b.readUInt32BE(0); h=b.readUInt32BE(4);
      assert.equal(b[8],8); assert(b[9]===2||b[9]===6);
      assert.deepEqual([...b.subarray(10)],[0,0,0]); channels=b[9]===2?3:4;
    }
    if(tag==='IDAT') parts.push(b);
    pos+=n+12;
  }
  const im=canvas(w,h), raw=zlib.inflateSync(Buffer.concat(parts)), decoded=Buffer.alloc(w*h*channels);
  for(let y=0;y<h;y++) {
    const f=raw[y*(w*channels+1)]; assert(f<=4);
    for(let x=0;x<w*channels;x++) {
      const a=x>=channels?decoded[y*w*channels+x-channels]:0;
      const b=y?decoded[(y-1)*w*channels+x]:0;
      const c=y&&x>=channels?decoded[(y-1)*w*channels+x-channels]:0, p=a+b-c;
      const ds=[Math.abs(p-a),Math.abs(p-b),Math.abs(p-c)];
      const pred=[0,a,b,Math.floor((a+b)/2),[a,b,c][ds.indexOf(Math.min(...ds))]][f];
      decoded[y*w*channels+x]=(raw[y*(w*channels+1)+x+1]+pred)&255;
    }
  }
  for(let i=0;i<w*h;i++) im.p.set(channels===4?decoded.subarray(i*4,i*4+4):[...decoded.subarray(i*3,i*3+3),255],i*4);
  return im;
}
function encodePNG(im) {
  const ih=Buffer.alloc(13), rows=Buffer.alloc((im.w*4+1)*im.h);
  ih.writeUInt32BE(im.w); ih.writeUInt32BE(im.h,4); ih[8]=8; ih[9]=6;
  for(let y=0;y<im.h;y++) im.p.copy(rows,y*(im.w*4+1)+1,y*im.w*4,(y+1)*im.w*4);
  return Buffer.concat([Buffer.from('89504e470d0a1a0a','hex'),chunk('IHDR',ih),
    chunk('IDAT',zlib.deflateSync(rows,{level:9})),chunk('IEND',Buffer.alloc(0))]);
}
function get(im,x,y) { return [...im.p.subarray((y*im.w+x)*4,(y*im.w+x)*4+4)]; }
function put(im,x,y,color) {
  assert(Number.isInteger(x)&&Number.isInteger(y)&&x>=0&&x<im.w&&y>=0&&y<im.h);
  im.p.set(color,(y*im.w+x)*4);
}
function rect(im,x,y,w,h,color) {
  for(let yy=y;yy<y+h;yy++) for(let xx=x;xx<x+w;xx++) put(im,xx,yy,color);
}
function line(im,points,color) {
  for(let i=1;i<points.length;i++) {
    let [x,y]=points[i-1]; const [bx,by]=points[i];
    const dx=Math.abs(bx-x),sx=x<bx?1:-1,dy=-Math.abs(by-y),sy=y<by?1:-1;
    let err=dx+dy;
    while(true) {
      put(im,x,y,color); if(x===bx&&y===by) break;
      const t=2*err; if(t>=dy){err+=dy;x+=sx;} if(t<=dx){err+=dx;y+=sy;}
    }
  }
}
function paste(dst,src,dx,dy) {
  for(let y=0;y<src.h;y++) for(let x=0;x<src.w;x++) {
    const c=get(src,x,y); if(c[3]) put(dst,x+dx,y+dy,c);
  }
}
function enlarge(im,n) {
  const big=canvas(im.w*n,im.h*n);
  for(let y=0;y<big.h;y++) for(let x=0;x<big.w;x++) put(big,x,y,get(im,Math.floor(x/n),Math.floor(y/n)));
  return big;
}

  }
  return [...counts.entries()].sort((a,b)=>b[1]-a[1]);
}

// Two explicit extra colours preserve the carved wood and muted brass without
// promoting the palette's bright gold to the dominant material. No other extras.
const EXTRA = ['4b3631','837044']; // dark carved wood; aged brass
const C=Object.fromEntries(Object.entries({
  ink:'080b14', deep:'1a1d33', navy:'2d3050', blue:'3f4a6e',
  metal:'62677d', paper:'d7b4a0', red:'7f1a22', redShade:'7a3a3c',
  teal:'226076', cyan:'42a9bd', green:'46705f', ochre:'b8862e',
  wood:EXTRA[0], brass:EXTRA[1]
}).map(([name,hex])=>[name,rgba(hex)]));
const allowed=Object.values(C);

function nearest(c) {
  let best,dist=Infinity;
  for(const p of allowed) {
    const d=(c[0]-p[0])**2+(c[1]-p[1])**2+(c[2]-p[2])**2;
    if(d<dist){dist=d;best=p;}
  }
  return best;
}
function background(c) {
  // Measured original backdrop is mostly #0a1124 / #0b1225, not exact #0b1026.
  // Flood only connected backdrop pixels, preserving enclosed dark materials.
  const [r,g,b]=c;
  return r<=23 && g<=29 && b>=24 && b<=49 && b-g>=5 && g-r>=-3;
}
function cutout(bounds) {
  const [x0,y0,x1,y1]=bounds, im=canvas(x1-x0,y1-y0);
  for(let y=0;y<im.h;y++) for(let x=0;x<im.w;x++) put(im,x,y,get(source,x+x0,y+y0));
  const seen=new Uint8Array(im.w*im.h),queue=[];
  function add(x,y) {
    if(x<0||y<0||x>=im.w||y>=im.h) return;
    const i=y*im.w+x;
    if(!seen[i]&&background(get(im,x,y))){seen[i]=1;queue.push(i);}
  }
  for(let x=0;x<im.w;x++){add(x,0);add(x,im.h-1);}
  for(let y=0;y<im.h;y++){add(0,y);add(im.w-1,y);}
  for(let q=0;q<queue.length;q++) {
    const i=queue[q],x=i%im.w,y=Math.floor(i/im.w);
    add(x-1,y);add(x+1,y);add(x,y-1);add(x,y+1);
  }
  for(let i=0;i<seen.length;i++) if(seen[i]) im.p.fill(0,i*4,i*4+4);
  removeSmallComponents(im,50);
  const box=bbox(im);
  const trimmed=canvas(box[2]-box[0]+1,box[3]-box[1]+1);
  for(let y=0;y<trimmed.h;y++) for(let x=0;x<trimmed.w;x++) put(trimmed,x,y,get(im,x+box[0],y+box[1]));
  return {im:trimmed,origin:[x0+box[0],y0+box[1]]};
}
function removeSmallComponents(im,min) {
  const seen=new Uint8Array(im.w*im.h);
  for(let start=0;start<seen.length;start++) if(!seen[start]&&im.p[start*4+3]) {
    const group=[start];seen[start]=1;
    for(let j=0;j<group.length;j++) {
      const i=group[j],x=i%im.w,y=Math.floor(i/im.w);
      for(const [xx,yy] of [[x-1,y],[x+1,y],[x,y-1],[x,y+1]]) {
        const k=yy*im.w+xx;
        if(xx>=0&&yy>=0&&xx<im.w&&yy<im.h&&!seen[k]&&im.p[k*4+3]){seen[k]=1;group.push(k);}
      }
    }
    if(group.length<min) for(const i of group) im.p.fill(0,i*4,i*4+4);
  }
}
function bbox(im) {
  let x0=im.w,y0=im.h,x1=-1,y1=-1;
  for(let y=0;y<im.h;y++) for(let x=0;x<im.w;x++) if(im.p[(y*im.w+x)*4+3]) {
    x0=Math.min(x0,x);y0=Math.min(y0,y);x1=Math.max(x1,x);y1=Math.max(y1,y);
  }
  assert(x1>=0,'Empty cutout');return [x0,y0,x1,y1];
}
function dominantFit(cut,w,h) {
  const src=cut.im, scale=Math.min(w/src.w,h/src.h);
  const fw=Math.round(src.w*scale),fh=Math.round(src.h*scale);
  const dx=Math.floor((w-fw)/2),dy=h-fh,out=canvas(w,h);
  for(let y=0;y<fh;y++) for(let x=0;x<fw;x++) {
    const xa=Math.floor(x*src.w/fw),xb=Math.floor((x+1)*src.w/fw);
    const ya=Math.floor(y*src.h/fh),yb=Math.floor((y+1)*src.h/fh);
    const bins=new Map();let opaque=0;
    for(let yy=ya;yy<yb;yy++) for(let xx=xa;xx<xb;xx++) {
      const c=get(src,xx,yy);if(!c[3])continue;opaque++;
      // RGB 5-bit bins absorb source antialias variations; vote BEFORE mapping.
      const key=((c[0]>>3)<<10)|((c[1]>>3)<<5)|(c[2]>>3);
      if(!bins.has(key)) bins.set(key,{n:0,colors:new Map()});
      const bin=bins.get(key),rgb=c.slice(0,3).join(',');
      bin.n++;bin.colors.set(rgb,(bin.colors.get(rgb)||0)+1);
    }
    if(opaque<0.5*(xb-xa)*(yb-ya))continue;
    const bin=[...bins.values()].sort((a,b)=>b.n-a.n)[0];
    const rgb=[...bin.colors].sort((a,b)=>b[1]-a[1])[0][0].split(',').map(Number);
    put(out,x+dx,y+dy,nearest(rgb));
  }
  removeSmallComponents(out,3);
  console.log('Cutout / fit:',cut.origin,[src.w,src.h],'->',[fw,fh],'offset',[dx,dy]);
  return out;
}
function outline(im) {
  const copy=Buffer.from(im.p);
  for(let y=0;y<im.h;y++) for(let x=0;x<im.w;x++) if(copy[(y*im.w+x)*4+3]) {
    if([[x-1,y],[x+1,y],[x,y-1],[x,y+1]].some(([xx,yy])=>
      xx<0||yy<0||xx>=im.w||yy>=im.h||!copy[(yy*im.w+xx)*4+3])) put(im,x,y,C.ink);
  }
}

function poly(im,points,color) {
  for(let y=Math.min(...points.map(p=>p[1]));y<=Math.max(...points.map(p=>p[1]));y++)
    for(let x=Math.min(...points.map(p=>p[0]));x<=Math.max(...points.map(p=>p[0]));x++) {
      let inside=false;
      for(let i=0,j=points.length-1;i<points.length;j=i++) {
        const [ax,ay]=points[i],[bx,by]=points[j];
        if((ay>y+.5)!==(by>y+.5)&&x+.5<(bx-ax)*(y+.5-ay)/(by-ay)+ax)inside=!inside;
      }
      if(inside)put(im,x,y,color);
    }
  line(im,[...points,points[0]],color);
}

function simplify(im) {
  // Merge isolated colour noise; preserve silhouette and deliberate accents
  // by doing this before the coordinate-specific hand cleanup below.
  const original={...im,p:Buffer.from(im.p)};
  for(let y=1;y<im.h-1;y++) for(let x=1;x<im.w-1;x++) {
    const c=get(original,x,y);if(!c[3])continue;
    const neighbors=[];
    for(let yy=y-1;yy<=y+1;yy++) for(let xx=x-1;xx<=x+1;xx++)
      if(xx!==x||yy!==y){const n=get(original,xx,yy);if(n[3])neighbors.push(n);}
    if(neighbors.length!==8||neighbors.some(n=>n.every((v,i)=>v===c[i])))continue;
    const counts=new Map();
    for(const n of neighbors){const k=n.join(',');counts.set(k,(counts.get(k)||0)+1);}
    const [key,n]=[...counts].sort((a,b)=>b[1]-a[1])[0];
    if(n>=4)put(im,x,y,key.split(',').map(Number));
  }
}
function cleanRack(im) {

    const c=get(im,x,y);
    if(c[3]&&(x<8||x>39||y<10||y>87)) {

      if(c.every((v,i)=>v===C.metal[i])||c.every((v,i)=>v===C.blue[i]))put(im,x,y,C.wood);

function cleanCables(im) {
  // Source registration: (649,341), 332x180 -> 74x40 at (3,0).
  // The Gemini sparkle at source (~935,485) occupies native (~67,32).
  // Replace the entire cable region containing it with explicit loop paths;
