  class Pic {
    public int W,H; public int[] P;
    public Pic(int w,int h,int c=0){W=w;H=h;P=Enumerable.Repeat(c,w*h).ToArray();}
    public int Get(int x,int y){return P[Math.Max(0,Math.Min(H-1,y))*W+Math.Max(0,Math.Min(W-1,x))];}
    public void Put(int x,int y,int c){if(x>=0&&y>=0&&x<W&&y<H)P[y*W+x]=c;}
    public Pic Copy(){var p=new Pic(W,H);Array.Copy(P,p.P,P.Length);return p;}
    public static Pic Load(string path){using(var b=new Bitmap(path)){var p=new Pic(b.Width,b.Height);for(int y=0;y<p.H;y++)for(int x=0;x<p.W;x++)p.Put(x,y,b.GetPixel(x,y).ToArgb());return p;}}
    public void Save(string path){using(var b=new Bitmap(W,H,PixelFormat.Format32bppArgb)){for(int y=0;y<H;y++)for(int x=0;x<W;x++)b.SetPixel(x,y,Color.FromArgb(Get(x,y)));b.Save(path,ImageFormat.Png);}}
    public void Rect(int x,int y,int w,int h,int c){for(int j=y;j<y+h;j++)for(int i=x;i<x+w;i++)Put(i,j,c);}
    public void Line(int x,int y,int xx,int yy,int c){int dx=Math.Abs(xx-x),sx=x<xx?1:-1,dy=-Math.Abs(yy-y),sy=y<yy?1:-1,e=dx+dy;while(true){Put(x,y,c);if(x==xx&&y==yy)break;int e2=2*e;if(e2>=dy){e+=dy;x+=sx;}if(e2<=dx){e+=dx;y+=sy;}}}
    public void Poly(int c,params int[] xy){int n=xy.Length/2;int lo=H,hi=0;for(int i=0;i<n;i++){lo=Math.Min(lo,xy[2*i+1]);hi=Math.Max(hi,xy[2*i+1]);}for(int y=lo;y<=hi;y++)for(int x=0;x<W;x++){bool inside=false;for(int i=0,j=n-1;i<n;j=i++){int xi=xy[2*i],yi=xy[2*i+1],xj=xy[2*j],yj=xy[2*j+1];if((yi>y)!=(yj>y)&&x<(double)(xj-xi)*(y-yi)/(yj-yi)+xi)inside=!inside;}if(inside)Put(x,y,c);}for(int i=0;i<n;i++)Line(xy[2*i],xy[2*i+1],xy[2*((i+1)%n)],xy[2*((i+1)%n)+1],c);}

  static int Bucket(int c){return ((c>>12)&0xf00)|((c>>8)&0xf0)|((c>>4)&15);}
  static Pic Vote(Pic src,int[] pal){
    // Source has no recoverable uniform grid: raster cells vary ~3-7 source px.
    // Impose the requested 320x180 grid. Extend 2 rows at each edge to 1024x576
    // before sampling, giving square 3.2x3.2-source-pixel cells without stretch.
    var p=new Pic(320,180);
    for(int y=0;y<180;y++)for(int x=0;x<320;x++){
      int x0=(int)Math.Floor(x*3.2),x1=(int)Math.Floor((x+1)*3.2);
      int y0=(int)Math.Floor(y*3.2)-2,y1=(int)Math.Floor((y+1)*3.2)-2;
      var bins=new Dictionary<int,int>();var exact=new Dictionary<int,int>();
      for(int j=y0;j<y1;j++)for(int i=x0;i<x1;i++){int c=src.Get(i,j),k=Bucket(c);bins[k]=bins.ContainsKey(k)?bins[k]+1:1;exact[c]=exact.ContainsKey(c)?exact[c]+1:1;}
      int center=src.Get((x0+x1-1)/2,(y0+y1-1)/2);
      int winner=bins.OrderByDescending(k=>k.Value).ThenBy(k=>k.Key==Bucket(center)?0:1).ThenBy(k=>k.Key).First().Key;
      // Actual source sample from the winning colour family, never a blended colour.
      int dominant=exact.Where(k=>Bucket(k.Key)==winner).OrderByDescending(k=>k.Value).ThenBy(k=>Dist(k.Key,center)).ThenBy(k=>k.Key).First().Key;
      p.Put(x,y,Nearest(dominant,pal));
    }return p;
  }
  static void CleanOrphans(Pic p){
    // Background only. Synchronous eight-neighbour vote removes single-pixel islands.
    // No filtering of the real sprites, lettering, broadcast marks, or cleaned outlines.
    for(int pass=0;pass<2;pass++){int[] old=(int[])p.P.Clone();for(int y=1;y<p.H-1;y++)for(int x=1;x<p.W-1;x++){
      int c=old[y*p.W+x];var count=new Dictionary<int,int>();for(int j=-1;j<=1;j++)for(int i=-1;i<=1;i++){if(i==0&&j==0)continue;int k=old[(y+j)*p.W+x+i];count[k]=count.ContainsKey(k)?count[k]+1:1;}
      if(!count.ContainsKey(c)){var win=count.OrderByDescending(k=>k.Value).ThenBy(k=>Dist(k.Key,c)).First();if(win.Value>=3)p.Put(x,y,win.Key);}
    }}
  }
  static void Simplify(Pic p){
    // Remove high-frequency conversion chatter before the authored pixel pass.
    for(int pass=0;pass<2;pass++){int[] old=(int[])p.P.Clone();for(int y=1;y<p.H-1;y++)for(int x=1;x<p.W-1;x++){
      var count=new Dictionary<int,int>();for(int j=-1;j<=1;j++)for(int i=-1;i<=1;i++){int k=old[(y+j)*p.W+x+i];count[k]=count.ContainsKey(k)?count[k]+1:1;}
      var win=count.OrderByDescending(k=>k.Value).ThenBy(k=>Dist(k.Key,old[y*p.W+x])).First();
      if(win.Value>=5&&Dist(win.Key,old[y*p.W+x])<14000)p.Put(x,y,win.Key);
    }}
  }

  static void CheckSprite(Pic sprite,Pic final,int ox,int oy){for(int y=0;y<sprite.H;y++)for(int x=0;x<sprite.W;x++)if((uint)sprite.Get(x,y)>>24==255)Check(sprite.Get(x,y)==final.Get(ox+x,oy+y),"Sprite changed or occluded");}
