    // Impose the requested 320x180 grid. Extend 2 rows at each edge to 1024x576
    // before sampling, giving square 3.2x3.2-source-pixel cells without stretch.
    var p=new Pic(320,180);
    for(int y=0;y<180;y++)for(int x=0;x<320;x++){
      int x0=(int)Math.Floor(x*3.2),x1=(int)Math.Floor((x+1)*3.2);
      int y0=(int)Math.Floor(y*3.2)-2,y1=(int)Math.Floor((y+1)*3.2)-2;
      var bins=new Dictionary<int,int>();var exact=new Dictionary<int,int>();
      for(int j=y0;j<y1;j++)for(int i=x0;i<x1;i++){int c=src.Get(i,j),k=Bucket(c);bins[k]=bins.ContainsKey(k)?bins[k]+1:1;exact[c]=exact.ContainsKey(c)?exact[c]+1:1;}
      int center=src.Get((x0+x1-1)/2,(y0+y1-1)/2);
      int winner=bins.OrderByDescending(k=>k.Value).ThenBy(k=>k.Key==Bucket(center)?0:1).ThenBy(k=>k.Key).First().Key;

      int dominant=exact.Where(k=>Bucket(k.Key)==winner).OrderByDescending(k=>k.Value).ThenBy(k=>Dist(k.Key,center)).ThenBy(k=>k.Key).First().Key;
      p.Put(x,y,Nearest(dominant,pal));
    }return p;

  public static void Run(string root){
    signCount=0;
    string dir=Path.Combine(root,"docs/design/concept/ending");
    var source=Pic.Load(Path.Combine(dir,"ending_v5_gemini.png"));
    Check(source.W==1024&&source.H==572,"Expected approved 1024x572 source");
    int[] project=Pic.Load(Path.Combine(root,"Assets/Art/Palette/number_palette.png")).P.Distinct().ToArray();
    int[] extra=new string[]{"345B80","4B7591","7D8C97","BEA08E","E3AB7B","F5BE82","FFF0BD","DE6BAF"}.Select(C).ToArray();
    int[] palette=project.Concat(extra).Distinct().ToArray();
    var reduced=Vote(source,palette);Simplify(reduced);CleanOrphans(reduced);
    var plate=reduced.Copy();Scene(plate);Hero(plate,reduced);
    // Remove accidental single-pixel islands after authored line intersections.
    // All intentional small marks (paper stripes, hilt, hair ties, rims) form
    // connected 2px-or-longer clusters; no star or sparkle is preserved.
    CleanOrphans(plate);
    Check(signCount==6,"Expected exactly six mounted signs");
    Check(plate.P.All(c=>palette.Contains(c)),"Pixel escaped the 32-colour palette");
    Check(plate.P.All(c=>((uint)c>>24)==255),"Non-opaque pixel");
    plate.Save(Path.Combine(dir,"ending_v5_320x180.png"));
    plate.Scale(3).Save(Path.Combine(dir,"ending_v5_960x540.png"));
    // Side-by-side at equal review scale, no labels/text. Source is DOWNscaled
    // to 960x536 and extended by 2 rows top/bottom (1024x572 -> 1024x576).
    var comparison=new Pic(1920,540);
    for(int y=0;y<540;y++)for(int x=0;x<960;x++)comparison.Put(x,y,source.Get(x*1024/960,(y*576/540)-2));
    comparison.Blit(plate.Scale(3),960,0);comparison.Save(Path.Combine(dir,"ending_v5_compare.png"));
